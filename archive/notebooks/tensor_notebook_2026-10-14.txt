CTD_NOAA_WHOI_KAUST$cast_id<-CTD_NOAA_WHOI_KAUST$DeepCTDID
CTD_NOAA_WHOI_KAUST$Region<-CTD_NOAA_WHOI_KAUST$Region.x
CTD_NOAA_WHOI_KAUST$date<-CTD_NOAA_WHOI_KAUST$UTCDate.x
#CTD_NOAA_WHOI_KAUST$lat<-CTD_NOAA_WHOI_KAUST$Latitude
#CTD_NOAA_WHOI_KAUST$lon<-CTD_NOAA_WHOI_KAUST$Longitude
CTD_NOAA_WHOI_KAUST$Depth_m<-CTD_NOAA_WHOI_KAUST$Depth_m
CTD_NOAA_WHOI_KAUST$Chl<-CTD_NOAA_WHOI_KAUST$Chl
CTD_NOAA_WHOI_KAUST$depth<-CTD_NOAA_WHOI_KAUST$Depth_m
CTD_NOAA_WHOI_KAUST$temperature<-CTD_NOAA_WHOI_KAUST$Temp_DegC

#x=38.9375, y=22.2751

CTD_NOAA_WHOI_KAUST <- CTD_NOAA_WHOI_KAUST %>%
  mutate(
    lat = ifelse(Region == "KAUST", 22.2751, lat),   # replace with your KAUST latitude
    lon = ifelse(Region == "KAUST", 38.9375, lon)    # replace with your KAUST longitude
  )

CTD_NOAA_WHOI_KAUST <- CTD_NOAA_WHOI_KAUST %>%
  mutate(
    distance = ifelse(Region == "KAUST", 2, distance)    # replace with your KAUST longitude
  )

CTD_NOAA_WHOI_KAUST_model<-CTD_NOAA_WHOI_KAUST%>%
  dplyr::select(LocationCode, cast_id, Region, date, lat, lon, Depth_m, Chl, depth, temperature, min_dist_m, distance)


CTD_NOAA_WHOI_KAUST_model <- CTD_NOAA_WHOI_KAUST_model %>%
  group_by(cast_id) %>%
  mutate(
    surface_chl = median(Chl[Depth_m >= 5 & Depth_m <= 15], na.rm = TRUE),
    chl_norm = ((Chl - surface_chl) / surface_chl) , #* 100 add back if we want % not *surface
    year = year(ymd(date)),
    log10_distance = log10(distance)
  ) %>%
  ungroup()


CTD_NOAA_WHOI_KAUST_model <- CTD_NOAA_WHOI_KAUST_model %>%
  mutate(
    enso = case_when(
      year %in% c(2009) ~ "el_nino",
      year %in% c(2010, 2011 ) ~ "la_nina",
      TRUE ~ "neutral"
    )
  )


str(CTD_NOAA_WHOI_KAUST_model)

CTD_binned <- CTD_NOAA_WHOI_KAUST_model %>%
  # create 5 m bins starting at 10 m
  mutate(depth_bin = cut(
    depth,
    breaks = seq(10, max(depth, na.rm = TRUE) + 10, by = 5), #average in 5m depth bins
    include.lowest = TRUE
  )) %>%
  # group by cast and depth bin
  group_by(cast_id, depth_bin) %>%
  summarise(
    Chl = median(Chl, na.rm = TRUE),          # median chl in 5 m bin
    depth = max(depth, na.rm = TRUE),          # upper bound (max) of the bin
    Region = first(Region),
    date = first(date),
    lat = first(lat),
    lon = first(lon),
    min_dist_m = first(min_dist_m),
    distance = first(distance),
    surface_chl = first(surface_chl),
    chl_norm = median(chl_norm, na.rm = TRUE),   # smoothed chl_norm too
    year = first(year),
    log10_distance = first(log10_distance),
    enso = first(enso),
    .groups = "drop"

  hdi_vals <- hdi(x, credMass = 0.95)

  c(
    mean   = mean(x),
    median = median(x),
    sd     = sd(x),
    se     = sd(x) / sqrt(length(x)),
    
    # quantile CI
    q2.5   = quantile(x, 0.025),
    q97.5  = quantile(x, 0.975),

    # HDI CI
    hdi_lwr = hdi_vals[1],
    hdi_upr = hdi_vals[2],

    # mean ± SD
    sd_lwr = mean(x) - sd(x),
    sd_upr = mean(x) + sd(x),

    # mean ± 1.96 SD
    norm_lwr = mean(x) - 1.96 * sd(x),

# We'll compute per-draw DCM depth, and two diffuseness metrics:
#  - FWHM: Full Width at Half Maximum (depth range where predicted >= half of draw's max)
#  - weighted_sd: weighted standard deviation of depth using predicted values as weights

# helper function to compute metrics for one region across depths
compute_metrics_for_region <- function(pred_matrix_region, depths) {
  # pred_matrix_region: matrix n_draws x n_depths (rows = draws)
  # depths: numeric vector of length n_depths
  n_draws <- nrow(pred_matrix_region)
  res <- tibble(
    draw = seq_len(n_draws),
    dcm_depth = numeric(n_draws),
    fwhm = numeric(n_draws),
    weighted_mean = numeric(n_draws),
    weighted_sd = numeric(n_draws),
    max_val = numeric(n_draws)
  )

  for(i in seq_len(n_draws)) {
    v <- pred_matrix_region[i, ]
    # handle NA or degenerate
    if(all(is.na(v))) {
      res$dcm_depth[i] <- NA
      res$fwhm[i] <- NA
      res$weighted_mean[i] <- NA
      res$weighted_sd[i] <- NA
      res$max_val[i] <- NA
      next
    }
    # DCM depth: depth where v is maximal (if multiple, take mean depth of ties)
    max_val <- max(v, na.rm = TRUE)
    res$max_val[i] <- max_val
    tie_idxs <- which(v >= (max_val - .Machine$double.eps*100))  # handle floats
    res$dcm_depth[i] <- mean(depths[tie_idxs])

    # FWHM: depths where v >= half max
    half <- max_val / 2
    idxs_half <- which(v >= half)
    if(length(idxs_half) == 0) {
      res$fwhm[i] <- 0
    } else {
      res$fwhm[i] <- max(depths[idxs_half]) - min(depths[idxs_half])
    }

    # weighted mean & sd (weights = predicted values, add small epsilon so zero weights ok)
    w <- pmax(v, 0)  # ensure non-negative
    if(sum(w, na.rm = TRUE) <= 0) {
      res$weighted_mean[i] <- NA
      res$weighted_sd[i] <- NA
    } else {
      wm <- sum(depths * w, na.rm = TRUE) / sum(w, na.rm = TRUE)
      res$weighted_mean[i] <- wm
      # weighted sd
      ws2 <- sum(w * (depths - wm)^2, na.rm = TRUE) / sum(w, na.rm = TRUE)
      res$weighted_sd[i] <- sqrt(ws2)
    }
  }
  res
}

#back calculate Ocean effects from posterior

library(brms)
library(dplyr)
library(tidyr)
library(ggplot2)

# 1. Define a grid of depths (adjust range and spacing to your study)
depth_grid <- seq(10, 250, by = 1)  # example: 0–200 m

# 2. Build prediction data for each region
# (Use representative or median values for covariates like log10_distance and enso)
newdata_region <- expand.grid(
  depth = depth_grid,
  log10_distance = median(CTD$log10_distance, na.rm = TRUE),
  enso = unique(CTD$enso, na.rm = TRUE),
  Region = unique(CTD$Region) #,
# date = unique(CTD$date)
)

# Add ocean for each region (use your actual lookup table if not in dataframe)
region_to_ocean <- CTD %>%
  distinct(Region, Ocean)
newdata_region <- left_join(newdata_region, region_to_ocean, by = "Region")

# 3. Get posterior draws for region-level predictions
#    re_formula = NULL includes region-level effects (important!)
posterior_region <- posterior_epred(
  fit_fixed_inter,        # your fitted brms model
  newdata = newdata_region,
  re_formula = NA  # include all group-level effects
)

# posterior_region is [iterations x nrow(newdata_region)] matrix

# 4. Reshape posterior predictions and average by Ocean
posterior_df <- as_draws_df(posterior_region) %>%
  mutate(draw = row_number()) %>%
  pivot_longer(
    cols = starts_with("V"),
    names_to = "row_id",
    values_to = "chl_pred"
  ) %>%
  mutate(row_id = as.integer(gsub("V", "", row_id))) %>%
  left_join(
    newdata_region %>%
      mutate(row_id = row_number()) %>%
      select(row_id, Region, Ocean, depth),
    by = "row_id"
  )

# 5. Aggregate across regions within each ocean (posterior mean at each draw)
posterior_ocean <- posterior_df %>%
  group_by(draw, Ocean, depth) %>%
  summarise(chl_pred = mean(chl_pred), .groups = "drop")

# 6. Summarize posterior mean and credible intervals
ocean_summary <- posterior_ocean %>%
  group_by(Ocean, depth) %>%
  summarise(
    mean = mean(chl_pred),
    lower = quantile(chl_pred, 0.025),
    upper = quantile(chl_pred, 0.975),
    .groups = "drop"
  )

# 7. Plot the ocean-level depth–chl curves
ggplot(ocean_summary, aes(x = depth, y = mean, color = Ocean, fill = Ocean)) +
  geom_line(size = 1) +
  geom_ribbon(aes(ymin = lower, ymax = upper), alpha = 0.25, color = NA) +
  scale_x_reverse() +  # depth usually increases downward
  coord_flip() +       # optional for oceanographic style
  labs(
    x = "Depth (m)",
    y = "Predicted normalized chlorophyll",
    title = "Ocean-level posterior mean chlorophyll-depth profiles"
  ) +
  theme_minimal(base_size = 14)
